def should_be_separate_line(line: str) -> bool:
    patterns = [
        r'^[^:]{1,80}:',  # Linee di dialogo
        r'^\*[^*]+\*',    # Effetti sonori

        r'^\.{3}',        # Linee che iniziano con ...
    ]
